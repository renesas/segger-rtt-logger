# e2 studio's CMake build configuration only offers GCC toolchains and passes
# no toolchain file, so CMake would pick arm-none-eabi-gcc from PATH, which
# rejects the LLVM flags generated by RASC (e.g. --target=arm-none-eabi).
if(NOT CMAKE_TOOLCHAIN_FILE)
	set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_LIST_DIR}/cmake/llvm.cmake" CACHE FILEPATH "Toolchain file")
endif()

# ARM_LLVM_TOOLCHAIN_PATH is a Windows path with backslashes. CMake 3.27
# (bundled with e2 studio) writes it unescaped into CMakeCCompiler.cmake and
# then fails: "Invalid character escape '\U'". Normalise before project().
foreach(_tc_env ARM_LLVM_TOOLCHAIN_PATH ARM_TOOLCHAIN_PATH)
	if(DEFINED ENV{${_tc_env}})
		file(TO_CMAKE_PATH "$ENV{${_tc_env}}" _tc_path)
		set(ENV{${_tc_env}} "${_tc_path}")
	endif()
endforeach()

# ─────────────────────────────────────────────────────────────────────────────
# Segger RTT Logger integration  (added on top of the RASC-generated project)
# ─────────────────────────────────────────────────────────────────────────────
# Toolchain: LLVM (Arm Toolchain for Embedded) with picolibc — see cmake/llvm.cmake.
#
# GeneratedSrc.cmake above defines the executable target ${PROJECT_NAME}.elf
# and recursively globs ra/, ra_gen/ and src/, so the Segger RTT Logger sources are
# compiled DIRECTLY into that ELF (not as a static archive) with no extra
# target_sources() needed:
#   src/app/                  application use-case (segger_rtt_logger_app.c/.h)
#   src/rs_segger_rtt_logger/ Segger RTT Logger RS: RTT logic + picolibc stdin/stdout/stderr streams
#   src/segger_rtt/           official SEGGER RTT V8.58 engine
#
# Constraints honoured here:
#   - No UART, no FreeRTOS.
#   - SEGGER_RTT_printf.c and SEGGER_RTT_Syscalls_*.c are deliberately NOT part
#     of src/segger_rtt/: rs_segger_rtt_logger_picolibc.c is the only stdio binding.
#   - Do not link libsemihost / libdummyhost: they also define stdin/stdout/stderr.
#   - ra_gen/main.c owns main(); src/hal_entry.c calls segger_rtt_logger_app_run().

# Segger RTT Logger include directories. Added here (not in GeneratedSrc.cmake)
# because RASC rewrites that file on "Generate Project Content".
target_include_directories(${PROJECT_NAME}.elf PRIVATE
	${CMAKE_CURRENT_SOURCE_DIR}/src/segger_rtt
	${CMAKE_CURRENT_SOURCE_DIR}/src/rs_segger_rtt_logger
	${CMAKE_CURRENT_SOURCE_DIR}/src/app)

# picolibc start-up code and C++ ABI runtime (same libraries as the e2 studio
# LLVM project: -lcrt0 -lc++abi).
target_link_libraries(${PROJECT_NAME}.elf PRIVATE
	crt0
	c++abi)
